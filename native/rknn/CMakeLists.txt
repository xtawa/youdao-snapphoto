cmake_minimum_required(VERSION 3.15)
project(snapphoto_rknn LANGUAGES CXX)
option(SNAPPHOTO_HAVE_RKNN "Build against RK3562 RKNN SDK" OFF)
add_library(snapphoto_rknn STATIC rknn_enhancer.cpp)
target_compile_features(snapphoto_rknn PUBLIC cxx_std_17)
if(SNAPPHOTO_HAVE_RKNN)
  target_compile_definitions(snapphoto_rknn PUBLIC SNAPPHOTO_HAVE_RKNN=1)
  target_include_directories(snapphoto_rknn PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/rknn")
  target_link_libraries(snapphoto_rknn PUBLIC ${CMAKE_DL_LIBS})
endif()
