cmake_minimum_required(VERSION 3.15)
project(snapphoto_rknn LANGUAGES CXX)
option(SNAPPHOTO_HAVE_RKNN "Build against RK3562 RKNN SDK" OFF)
add_library(snapphoto_rknn STATIC rknn_enhancer.cpp)
target_compile_features(snapphoto_rknn PUBLIC cxx_std_17)
if(SNAPPHOTO_HAVE_RKNN)
  if(NOT DEFINED RKNN_SDK_INCLUDE OR NOT DEFINED RKNN_SDK_LIBRARY)
    message(FATAL_ERROR "Set RKNN_SDK_INCLUDE and RKNN_SDK_LIBRARY for X7 Pro")
  endif()
  target_compile_definitions(snapphoto_rknn PUBLIC SNAPPHOTO_HAVE_RKNN=1)
  target_include_directories(snapphoto_rknn PRIVATE "\${RKNN_SDK_INCLUDE}")
  target_link_libraries(snapphoto_rknn PUBLIC "\${RKNN_SDK_LIBRARY}")
endif()
